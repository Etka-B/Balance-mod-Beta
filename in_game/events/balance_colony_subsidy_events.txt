namespace = balance_colony_subsidy

# THE COLONY WINDOW'S DAILY RENEWAL (scripted_effects/
# balance_colony_effects.txt). The game cannot bar a subject from closing one
# building (can_close and ai_forbid_shutdown are yes/no on the whole type) nor
# from taking a subsidy away, so every day a locked building the subject closed
# is rebuilt open, and in "Keep open" locations every subsidy it took is given
# straight back. A country event on the overlord that re-arms
# itself every day, the same chain as balance_combat_probe.1 (which measured
# that days = 1 does re-fire daily). Started by balance_colony_subsidy_seed;
# ends by itself once the overlord has nothing left for it to do.

balance_colony_subsidy.1 = {
	type = country_event
	outcome = neutral
	title = empty_text
	desc = empty_text
	hidden = yes

	immediate = {
		set_variable = balance_subsidy_alive
		balance_colony_subsidy_renew = yes
		if = {
			limit = { balance_colony_daily_needed = yes }
			trigger_event_silently = {
				id = balance_colony_subsidy.1
				days = 1
			}
		}
		else = {
			remove_variable = balance_subsidy_running
		}
	}
}
