african_newland_developmentboost = {
    ability = adm
    icon = develop_province
    allow_multiple = yes
    
    potential = {
        original_capital ?= { continent = continent:africa }
    }
    
    select_trigger = {
        looking_for_a = province
        source = actor
        target_flag = target
        name = "swe_slash_and_burn_new_farms_select_province"
        none_available_msg_key = "african_newland_developmentboost_no_provinces"
        column = {
            data = name
        }
        column = {
            data = development
        }
        column = {
            data = population
        }
        visible = {
            province_average_development < 10
            continent = continent:africa       
            NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = african_newland_developmentboost interaction_target = target } }
            NOT = { cabinet_already_performing_same_task_on_target = { cabinet_action = develop_province interaction_target = target } }
        }
        map_color = {
            lerp = {
                min_color = define:NMapColors|MAP_COLOR_LOW
                max_color = define:NMapColors|MAP_COLOR_HIGH
                factor = province.province_average_development
            }
        }
    }

    progress = {
        scope:target = {
            value = province_average_development
            divide = 100
        }
    }

    is_finished = {
        scope:target ?= {
            province_average_development > 12
        }
    }
    
    location_modifier = {
        local_monthly_development = 0.0075
    }

    map_marker = {
        show_for_owner = yes
    }
}
