# THE COLONY WINDOW'S BUILDING LOCKS (gui/balance_colony_window.gui).
# An overlord marks a subject's location:
#   balance_keep_all        no building there may be taken down by the subject
#   balance_keep_buildings  a list of single buildings the subject may not take down
# Both lapse by themselves once the location is no longer a subject's.

# can_destroy (root = location, scope:building = the building)
balance_building_not_kept = {
	trigger_if = {
		limit = {
			# only the subject is held back: its overlord (or anyone else with a
			# building here) may still take its own building down
			exists = scope:actor
			owner ?= { is_subject = yes }
			owner ?= scope:actor
			OR = {
				has_variable = balance_keep_all		# the whole location, later buildings too
				AND = {
					exists = scope:building
					is_target_in_variable_list = { name = balance_keep_buildings target = scope:building }
				}
			}
		}
		custom_tooltip = {
			text = balance_keep_building_tt
			always = no
		}
	}
}

# building scope: types the lock does not reach (none: every type carries it,
# see write_building_locks.py); kept for the window's buttons
balance_building_lock_unsupported = {
	always = no
}

# building scope: one of the subject's own (scope:balance_colony_o) that it has
# closed, and not still going up: rebuilt open by balance_colony_rebuild_gathered
balance_colony_is_closed_reopenable = {
	owner ?= scope:balance_colony_o
	is_opened = no
	building_level >= 1
	NOT = { building_levels_under_construction >= 1 }
}

# country scope: whether the overlord needs the daily chain
# (events/balance_colony_subsidy_events.txt): "Keep open" locations, or
# building locks in its subjects' lands to reopen
balance_colony_daily_needed = {
	OR = {
		any_in_list = { variable = balance_subsidy_locations always = yes }
		AND = {
			has_variable = balance_colony_uses_locks
			any_subject = { always = yes }
		}
	}
}
