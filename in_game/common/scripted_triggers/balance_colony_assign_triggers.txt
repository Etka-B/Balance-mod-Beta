# scope:node has a rule of its own at its level (S N K A = the level's lists;
# root = the overlord)

	OR = {
		any_subject = { is_target_in_variable_list = { name = $S$ target = scope:node } }
		is_target_in_variable_list = { name = $N$ target = scope:node }
		is_target_in_variable_list = { name = $K$ target = scope:node }
		is_target_in_variable_list = { name = $A$ target = scope:node }
		is_target_in_variable_list = { name = $O$ target = scope:node }
	}
}
