# location scope (scope:balance_prio_t = the row's building, scope:balance_prio_c =
# the subject): an older level of it, the subject's own, stands here. Building it
# here replaces that one.
balance_prio_has_predecessor_here = {
	OR = {
		AND = {
			scope:balance_prio_t = building_type:armory
			any_buildings_in_location = {
				owner ?= scope:balance_prio_c
				OR = { building_type = building_type:korean_barracks building_type = building_type:provincial_garrison }
				building_level >= 1
			}
		}
		AND = {
			scope:balance_prio_t = building_type:arts_academy
			any_buildings_in_location = {
				owner ?= scope:balance_prio_c

		}
		AND = {
			scope:balance_prio_t = building_type:barracks
