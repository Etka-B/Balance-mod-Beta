# THE HUSSARS ARRIVE AN AGE EARLY HERE, SO THEY ARE BUILT ON THAT AGE.
# balance_fla/FLA_country.txt moves hun_hungarian_hussars to age_3_discovery,
# but the unit itself still copied a_age_4_reformation_light_cavalry and so kept
# that age's numbers: max_strength 0.8 where every other age 3 light cavalry has
# 0.6, and an age_4_reformation tag on a unit you can field in age 3. The two
# templates differ in nothing else -- combat_power is 4 in both -- so this is
# the vanilla unit with the age 3 template under it.
#
# NO combat_power LINE, AND THAT IS DELIBERATE. A unit's combat_power is ADDED
# to the one it copies, and this file has already rewritten the templates
# themselves: age 3 cavalry is 2.75 here, age 4 is 3.75, where the game ships
# 4 for every age. The INJECT offsets further up were written against the
# game's 4 -- 4 minus 1.25 is the 2.75 the age 3 template now carries -- so on
# top of the rewritten template they land twice. Measured in game: with
# -1.25 written here the hussars showed 1.50, against 2.75 for a plain age 3
# light cavalry, which is why they stopped hitting. Nothing added, they take
# the template's 2.75, the same as any other age 3 light cavalry -- which is
# how the game itself builds its uniques: same power as their age, the edge
# coming from initiative, the noble mercenaries and the terrain, all below.
#
# (The same double count is sitting on a_akinji, a_lanzas_de_castilla and the
# rest of the INJECT list above; left alone, not asked for.)
TRY_REPLACE:a_hungarian_hussars = {
	copy_from = a_age_3_discovery_light_cavalry
	is_special = yes
	upgrades_to = a_hussars

	#root = location. Checks location_potential first so don't need that scripting in
	mercenaries_per_location = {
		pop_type = nobles
		multiply = 0.3
	}

	buildable = yes

	initiative = 2

	impact = {
		flatland = -0.25
		mountains = -0.25
	}

	gfx_tags = { medium_tag }
}
