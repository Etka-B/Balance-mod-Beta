# Overrides game/in_game/common/country_interactions/improve_cultural_view_subject.txt
#
# Only the cooldown is changed: 50 years -> 20. Everything else is the base
# game's file, copied whole, because a country_interaction is replaced by name
# rather than merged -- leaving anything out would delete it from the game.
#
# The base game's own diplomacy-side twin, improve_cultural_view, has no
# cooldown at all (it is paid for with diplomatic cost instead), so the subject
# version was the only place this number lives.

improve_cultural_view_subject = {
	type = subject
	category = CATEGORY_SUBJECT_ACTIONS
	
	
	potential = {
		scope:actor = {
			culture = {
				dominant_country ?= scope:actor
			}
		}
	}

	allow = {
	}
	
	ai_limit_per_check = 1 #Because it has the cooldown

	cooldown = {
		type = improve_cultural_view_subject
		years = 20
	}

	select_trigger = {
		looking_for_a = country
		target_flag = recipient
		name = "choose_country"
		column = {
			data = name
		}
		visible = {
			culture = {
				dominant_country ?= scope:recipient
			}
		}
        enabled = {
			NOT = {
				culture = {
					cultural_view = {
						target = scope:actor.culture
						value = kindred
					}
				}
			}
			subject_loyalty >= MIN_SUBJECT_LOYALTY_FOR_ACTION
			liberty_desire < MAX_SUBJECT_LIBERTY_DESIRE_FOR_ACTION
		}
	}

	effect = {
		scope:recipient = {
			culture = {
				change_cultural_view = {
					target = scope:actor.culture
					change = 1
				}
			}
			add_liberty_desire = liberty_desire_mild_plus
		}
		if = {
			limit = {
				scope:actor = {
					is_human = yes
				}
			}
			scope:actor = {
				set_variable = {
					name = unlock_achievement_hearts_and_minds
					months = 1
				}
			}
		}
	}

	ai_will_do = {
		# opinion of the other culture
		add = {
			if = {
				limit = {
					scope:recipient = {
						culture = {
							cultural_view = {
								target = scope:actor.culture
								value = enemy
							}
						}
					}
				}
				value = 45
			}
		}
		add = {
			if = {
				limit = {
					scope:recipient = {
						culture = {
							cultural_view = {
								target = scope:actor.culture
								value = negative
							}
						}
					}
				}
				value = 25
			}
		}
		add = {
			if = {
				limit = {
					scope:recipient = {
						culture = {
							cultural_view = {
								target = scope:actor.culture
								value = neutral
							}
						}
					}
				}
				value = 10
			}
		}
	}
}
