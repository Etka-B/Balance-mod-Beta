balance_tier_clear = {
	if = { limit = { has_variable = balance_tier_row } remove_variable = balance_tier_row }
	clear_variable_list = balance_tier_list
}

# subject scope: row S now holds scope:btype. Its profit readings were for the
# old building; "upgrades only" goes back to "everywhere" when the new one
# replaces nothing.
balance_tier_set_slot = {
	set_variable = { name = balance_prio_$S$_b value = scope:btype }
	if = { limit = { has_variable = balance_prio_$S$_pavg } remove_variable = balance_prio_$S$_pavg }
	if = { limit = { has_variable = balance_prio_$S$_pn } remove_variable = balance_prio_$S$_pn }
	if = {
		limit = {
			var:balance_prio_$S$_where ?= 3
			NOT = { balance_prio_row_can_upgrade = { S = $S$ } }
		}
		set_variable = { name = balance_prio_$S$_where value = 0 }
	}
}
