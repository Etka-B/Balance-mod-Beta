# ---- the building filter's icons (root = the overlord): every building type
# standing in the picked subject's locations, in two rows
balance_colony_fill_btypes = {
	clear_variable_list = balance_colony_btypes
	clear_variable_list = balance_colony_btypes2
	set_variable = { name = balance_colony_bn value = 0 }
	if = {
		limit = { has_variable = balance_colony_subject }
		var:balance_colony_subject = {
			every_owned_location = {
				every_buildings_in_location = {
					building_type = { save_temporary_scope_as = balance_colony_bt }
					root = {
						if = {
							limit = {
								NOT = { is_target_in_variable_list = { name = balance_colony_btypes target = scope:balance_colony_bt } }
								NOT = { is_target_in_variable_list = { name = balance_colony_btypes2 target = scope:balance_colony_bt } }
							}
							if = {
								limit = { var:balance_colony_bn < 20 }
								add_to_variable_list = { name = balance_colony_btypes target = scope:balance_colony_bt }
							}
							else = {
								add_to_variable_list = { name = balance_colony_btypes2 target = scope:balance_colony_bt }
							}
							change_variable = { name = balance_colony_bn add = 1 }
						}
					}
				}
			}
		}
	}
	remove_variable = balance_colony_bn
}

# ---- subsidies by building type (this = the location, root = the overlord,
# scope:balance_colony_st = the building type). balance_subsidy_types on the
# location lists the types, balance_subsidy_tby holds the overlord that set
# them; the daily renewal (balance_colony_subsidy_renew) keeps them up.
balance_colony_subsidize_types_here = {
	owner = { save_temporary_scope_as = balance_colony_o }
	save_temporary_scope_as = balance_colony_here
	every_in_list = {
		variable = balance_subsidy_types
		save_temporary_scope_as = balance_colony_st2
		scope:balance_colony_here = {
			every_buildings_in_location = {
				limit = {
					building_type = scope:balance_colony_st2
					owner ?= scope:balance_colony_o
					NOT = { is_subsidized = yes }
				}
				set_subsidized = yes
			}
		}
	}
}

balance_colony_subsidy_type_on_here = {
	if = {
		limit = { NOT = { is_target_in_variable_list = { name = balance_subsidy_types target = scope:balance_colony_st } } }
		add_to_variable_list = { name = balance_subsidy_types target = scope:balance_colony_st }
	}
	set_variable = { name = balance_subsidy_tby value = root }
	save_temporary_scope_as = balance_colony_sl
	root = {
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = balance_subsidy_locations target = scope:balance_colony_sl } } }
			add_to_variable_list = { name = balance_subsidy_locations target = scope:balance_colony_sl }
		}
	}
	balance_colony_subsidize_types_here = yes
	root = { balance_colony_subsidy_seed = yes }
}

balance_colony_subsidy_type_off_here = {
	remove_list_variable = { name = balance_subsidy_types target = scope:balance_colony_st }
	if = {
		limit = { NOT = { var:balance_subsidy_all ?= root } }
		owner = { save_temporary_scope_as = balance_colony_o }
		every_buildings_in_location = {
			limit = {
				building_type = scope:balance_colony_st
				owner ?= scope:balance_colony_o
				is_subsidized = yes
			}
			set_subsidized = no
		}
	}
	if = {
		limit = { NOT = { any_in_list = { variable = balance_subsidy_types always = yes } } }
		if = { limit = { has_variable = balance_subsidy_tby } remove_variable = balance_subsidy_tby }
		if = {
			limit = { NOT = { var:balance_subsidy_all ?= root } }
			save_temporary_scope_as = balance_colony_sl
			root = { remove_list_variable = { name = balance_subsidy_locations target = scope:balance_colony_sl } }
		}
	}
}

# ---- the priorities tab: an ordered list of up to 20 rows on the subject.
# Row S is the variables balance_prio_S_b (a building type) or _g (a raw
# material), _gold (build only while the treasury holds this much), _max (how
# many of it may be under construction at once), _where (0 everywhere,
# 1 the saved selection, 2 towns and cities), _hold (while this row still has
# work, the rows under it wait), _busy (under construction at the last look),
# _pmin (build only while its buildings of this type earn this much on
# average), _pavg (that average at the last look) and _mode (where first:
# 0 most populous, 1 most efficient, 2 most profitable / best price). Row 0 is
# scratch for swapping. The overlord keeps the subjects it gave orders to in
# balance_prio_subjects. Written by colony_v10.py and colony_v11.py.
balance_colony_prio_defaults = {
	if = { limit = { NOT = { has_variable = balance_prio_freq } } set_variable = { name = balance_prio_freq value = 1 } }
}

# this = the subject, root = the overlord: switch the orders on and remember it
balance_colony_prio_enable = {
	set_variable = balance_prio_on
	balance_colony_prio_defaults = yes
	save_temporary_scope_as = balance_prio_s
	root = {
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = balance_prio_subjects target = scope:balance_prio_s } } }
			add_to_variable_list = { name = balance_prio_subjects target = scope:balance_prio_s }
		}
	}
}

# ---- rows (this = the subject)
balance_prio_clear_slot = {
	if = { limit = { has_variable = balance_prio_$S$_b } remove_variable = balance_prio_$S$_b }
	if = { limit = { has_variable = balance_prio_$S$_g } remove_variable = balance_prio_$S$_g }
	if = { limit = { has_variable = balance_prio_$S$_gold } remove_variable = balance_prio_$S$_gold }
	if = { limit = { has_variable = balance_prio_$S$_max } remove_variable = balance_prio_$S$_max }
	if = { limit = { has_variable = balance_prio_$S$_where } remove_variable = balance_prio_$S$_where }
	if = { limit = { has_variable = balance_prio_$S$_busy } remove_variable = balance_prio_$S$_busy }
	if = { limit = { has_variable = balance_prio_$S$_pmin } remove_variable = balance_prio_$S$_pmin }
	if = { limit = { has_variable = balance_prio_$S$_pavg } remove_variable = balance_prio_$S$_pavg }
	if = { limit = { has_variable = balance_prio_$S$_pn } remove_variable = balance_prio_$S$_pn }
	if = { limit = { has_variable = balance_prio_$S$_mode } remove_variable = balance_prio_$S$_mode }
	if = { limit = { has_variable = balance_prio_$S$_cmin } remove_variable = balance_prio_$S$_cmin }

balance_prio_copy_slot = {
	balance_prio_clear_slot = { S = $TO$ }
	save_temporary_scope_as = balance_prio_cs
	every_in_list = {
		variable = balance_prio_$FROM$_locs
		save_temporary_scope_as = balance_prio_cl
		scope:balance_prio_cs = { add_to_variable_list = { name = balance_prio_$TO$_locs target = scope:balance_prio_cl } }
	}
	if = { limit = { has_variable = balance_prio_$FROM$_b } set_variable = { name = balance_prio_$TO$_b value = var:balance_prio_$FROM$_b } }
	if = { limit = { has_variable = balance_prio_$FROM$_g } set_variable = { name = balance_prio_$TO$_g value = var:balance_prio_$FROM$_g } }

# S = the row; sets balance_prio_busy and scope:balance_prio_t / _g
balance_prio_count_slot = {
	if = { limit = { NOT = { has_variable = balance_prio_$S$_mode } } set_variable = { name = balance_prio_$S$_mode value = 0 } }
	set_variable = { name = balance_prio_busy value = 0 }
	if = {
		limit = { has_variable = balance_prio_$S$_b }
		var:balance_prio_$S$_b = { save_temporary_scope_as = balance_prio_t }
		every_owned_location = {
			limit = {
				any_buildings_in_location = {
					building_type = scope:balance_prio_t
					building_levels_under_construction > 0
				}
			}
			scope:balance_prio_c = { change_variable = { name = balance_prio_busy add = 1 } }
		}
		# the average profit of its own buildings of this type
		set_variable = { name = balance_prio_n value = balance_sv_prio_t_count }
		if = {
			limit = { var:balance_prio_n > 0 }
			set_variable = { name = balance_prio_$S$_pavg value = balance_sv_prio_t_profit_sum }
			change_variable = { name = balance_prio_$S$_pavg divide = var:balance_prio_n }
			set_variable = { name = balance_prio_$S$_pn value = var:balance_prio_n }
		}
		if = {
			limit = {
				var:balance_prio_n <= 0

# one row: as many orders as its limit leaves room for, while the treasury
# holds its minimum and (buildings) its buildings of this type earn enough
balance_prio_run_slot = {
	balance_prio_count_slot = { S = $S$ }
	set_variable = { name = balance_prio_w value = var:balance_prio_$S$_where }
	set_variable = { name = balance_prio_m value = var:balance_prio_$S$_mode }
	# the row's own locations and control floor, for balance_prio_where_ok
	clear_variable_list = balance_prio_curlocs
	save_temporary_scope_as = balance_prio_cs
	every_in_list = {
		variable = balance_prio_$S$_locs
		save_temporary_scope_as = balance_prio_cl
		scope:balance_prio_cs = { add_to_variable_list = { name = balance_prio_curlocs target = scope:balance_prio_cl } }
	}
	if = {
		limit = { has_variable = balance_prio_$S$_cmin }
		set_variable = { name = balance_prio_cm value = var:balance_prio_$S$_cmin }
	}
	else = {
		set_variable = { name = balance_prio_cm value = 0 }
	}
	set_variable = { name = balance_prio_need value = var:balance_prio_$S$_gold }
	set_variable = { name = balance_prio_left value = var:balance_prio_$S$_max }
	change_variable = { name = balance_prio_left subtract = var:balance_prio_busy }
	if = { limit = { has_variable = balance_prio_pgate } remove_variable = balance_prio_pgate }
	if = {
		limit = {
			has_variable = balance_prio_$S$_pmin
			has_variable = balance_prio_$S$_pavg
			var:balance_prio_$S$_pavg < var:balance_prio_$S$_pmin
		}
		# they earn too little now: no more of them until the profit is back
		set_variable = balance_prio_pgate
		set_variable = { name = balance_prio_left value = 0 }
	}
	if = {
		limit = { has_variable = balance_prio_$S$_b }
		if = {
			limit = {
				can_build_building = scope:balance_prio_t
				NOT = { is_target_in_variable_list = { name = balance_forbid_buildings target = scope:balance_prio_t } }
			}
			while = {
				limit = {
					var:balance_prio_left > 0
					gold >= var:balance_prio_need
					any_owned_location = { balance_prio_b_place_ok = yes }
				}
				balance_prio_place_b = yes
				change_variable = { name = balance_prio_left add = -1 }
			}
		}
		if = {
			limit = {
				has_variable = balance_prio_$S$_hold
				OR = {
					var:balance_prio_busy > 0
					AND = {
						NOT = { has_variable = balance_prio_pgate }
						can_build_building = scope:balance_prio_t
						NOT = { is_target_in_variable_list = { name = balance_forbid_buildings target = scope:balance_prio_t } }
						any_owned_location = { balance_prio_b_place_ok = yes }
					}
				}
			}
			set_variable = balance_prio_stop
		}
	}
	if = {
		limit = { has_variable = balance_prio_$S$_g }
		while = {
			limit = {
				var:balance_prio_left > 0
				gold >= var:balance_prio_need
				any_owned_location = { balance_prio_g_place_ok = yes }
			}
			balance_prio_place_g = yes
			change_variable = { name = balance_prio_left add = -1 }
		}
		if = {
			limit = {
				has_variable = balance_prio_$S$_hold
				OR = {
					var:balance_prio_busy > 0
					any_owned_location = { balance_prio_g_place_ok = yes }
				}
			}
			set_variable = balance_prio_stop
		}
	}
	if = { limit = { has_variable = balance_prio_pgate } remove_variable = balance_prio_pgate }
	remove_variable = balance_prio_m
	remove_variable = balance_prio_cm
	clear_variable_list = balance_prio_curlocs
	remove_variable = balance_prio_w
	remove_variable = balance_prio_need
	remove_variable = balance_prio_left
	remove_variable = balance_prio_busy
}
