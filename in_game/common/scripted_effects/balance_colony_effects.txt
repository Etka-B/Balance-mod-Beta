# THE COLONY WINDOW'S LISTS AND LOCKS (gui/balance_colony_window.gui). The
# window cannot ask the game for a subject's locations or a location's
# buildings, so these fill variable lists on the player's country (root):
#   balance_colony_subjects   our subjects
#   balance_colony_locations  the picked subject's locations that pass the filters
#   balance_colony_goods      the raw materials of that subject's locations
#   balance_colony_buildings  the picked location's buildings
#   balance_colony_selected   locations picked for the bulk buttons

balance_colony_fill_subjects = {
	clear_variable_list = balance_colony_subjects
	# biggest first
	ordered_subject = {
		order_by = num_locations
		max = 1000
		check_range_bounds = no
		save_temporary_scope_as = balance_colony_s
		root = {
			add_to_variable_list = { name = balance_colony_subjects target = scope:balance_colony_s }
		}
	}
}

balance_colony_fill_locations = {
	clear_variable_list = balance_colony_locations
	clear_variable_list = balance_colony_goods
	if = {
		limit = { has_variable = balance_colony_subject }
		if = {
			limit = { has_variable = balance_colony_f_goods }
			var:balance_colony_f_goods = { save_temporary_scope_as = balance_colony_fg }
		}
		var:balance_colony_subject = {
			every_owned_location = {
				save_temporary_scope_as = balance_colony_l
				raw_material ?= {
					save_temporary_scope_as = balance_colony_g
					root = {
						if = {
							limit = { NOT = { is_target_in_variable_list = { name = balance_colony_goods target = scope:balance_colony_g } } }
							add_to_variable_list = { name = balance_colony_goods target = scope:balance_colony_g }
						}
					}
				}
				if = {
					limit = { balance_colony_location_shown = yes }
					root = {
						add_to_variable_list = { name = balance_colony_locations target = scope:balance_colony_l }
					}
				}
			}
		}
	}
}

balance_colony_fill_buildings = {
	clear_variable_list = balance_colony_buildings
	if = {
		limit = { has_variable = balance_colony_location }
		var:balance_colony_location = {
			every_buildings_in_location = {
				save_temporary_scope_as = balance_colony_b
				root = {
					add_to_variable_list = { name = balance_colony_buildings target = scope:balance_colony_b }
				}
			}
		}
	}
}

balance_colony_clear_filters = {
	if = { limit = { has_variable = balance_colony_f_rural } remove_variable = balance_colony_f_rural }
	if = { limit = { has_variable = balance_colony_f_town } remove_variable = balance_colony_f_town }
	if = { limit = { has_variable = balance_colony_f_city } remove_variable = balance_colony_f_city }
	if = { limit = { has_variable = balance_colony_f_goods } remove_variable = balance_colony_f_goods }
}

# ---- building locks on one location (this = the location, root = the overlord);
# balance_keep_all / balance_hold_all on the location mark a whole-location lock
balance_colony_keep_buildings_here = {
	set_variable = balance_keep_all
}

balance_colony_free_keep_here = {
	clear_variable_list = balance_keep_buildings
	if = { limit = { has_variable = balance_keep_all } remove_variable = balance_keep_all }
}

# ---- subsidies on one location (this = the location, root = the overlord).
# can_close is a plain yes/no on the building type and the game has no effect
# that reopens a closed building, so closing cannot be barred per location.
# What it closes over is the loss: a subsidised building has none (the
# location window drops its "unprofitable" mark), so balance_subsidy_all makes
# the subject subsidise its own buildings here. It holds the overlord that set
# it, so a subject handed to a new overlord keeps only the new one's. Renewed
# every day by balance_colony_subsidy_renew: that also reaches buildings built
# later, and undoes the subject's AI taking a subsidy away (nothing in the
# game stops it doing that, so the best we can do is give it straight back).
balance_colony_subsidize_here = {
	owner = { save_temporary_scope_as = balance_colony_o }
	every_buildings_in_location = {
		limit = {
			owner ?= scope:balance_colony_o		# never ours, nor a third country's: their owner pays
			NOT = { is_subsidized = yes }
		}
		set_subsidized = yes
	}
}

# ---- reopening. The game has no effect that opens a closed building, and
# its only bar on closing (ai_forbid_shutdown, can_close) is a yes/no on the
# whole building type, so it would reach every country. So a closed building
# we keep is taken down and the same type put back at the same level for the
# same owner, which comes up open. Its production method starts from the
# default and its staff are hired again; a lock on it moves to the new one.
# Buildings still going up are left alone. Done for the buildings the overlord
# locked (balance_colony_reopen_kept_here, every day from
# balance_colony_subsidy_renew) and in "Keep open" locations
# (balance_colony_reopen_here: a closed building cannot be subsidised either,
# a subsidy keeps staff from being laid off and a closed one has none).
# this = the location, root = the overlord.
balance_colony_reopen_here = {
	save_temporary_scope_as = balance_colony_here
	owner = { save_temporary_scope_as = balance_colony_o }
	root = { clear_variable_list = balance_colony_reopen }
	balance_colony_gather_closed_here = yes
	balance_colony_rebuild_gathered = yes
}

balance_colony_reopen_kept_here = {
	save_temporary_scope_as = balance_colony_here
	owner = { save_temporary_scope_as = balance_colony_o }
	root = { clear_variable_list = balance_colony_reopen }
	if = {
		limit = { has_variable = balance_keep_all }
		balance_colony_gather_closed_here = yes
	}
	else = {
		every_in_list = {
			variable = balance_keep_buildings
			limit = { balance_colony_is_closed_reopenable = yes }
			save_temporary_scope_as = balance_colony_b
			root = { add_to_variable_list = { name = balance_colony_reopen target = scope:balance_colony_b } }
		}
	}
	balance_colony_rebuild_gathered = yes
}

# gathered first: buildings are taken down and put up below, not while walking
# the location's own list
balance_colony_gather_closed_here = {
	every_buildings_in_location = {
		limit = { balance_colony_is_closed_reopenable = yes }
		save_temporary_scope_as = balance_colony_b
		root = { add_to_variable_list = { name = balance_colony_reopen target = scope:balance_colony_b } }
	}
}

balance_colony_rebuild_gathered = {
	root = {
		every_in_list = {
			variable = balance_colony_reopen
			save_temporary_scope_as = balance_colony_b
			building_type = { save_temporary_scope_as = balance_colony_t }
			if = {
				limit = { has_local_variable = balance_colony_was_kept }
				remove_local_variable = balance_colony_was_kept
			}
			if = {
				limit = {
					scope:balance_colony_here = {
						is_target_in_variable_list = { name = balance_keep_buildings target = scope:balance_colony_b }
					}
				}
				set_local_variable = { name = balance_colony_was_kept value = 1 }
				scope:balance_colony_here = {
					remove_list_variable = { name = balance_keep_buildings target = scope:balance_colony_b }
				}
			}
			# the game gives no level as a number: count it down to one, the
			# way balance_colony_free_hold_here moves levels
			set_local_variable = { name = balance_colony_lvl value = 1 }
			while = {
				limit = { scope:balance_colony_b = { building_level >= 2 } }
				scope:balance_colony_b = { change_building_level = -1 }
				change_local_variable = { name = balance_colony_lvl add = 1 }
			}
			scope:balance_colony_here = {
				destroy_building_forcefully = scope:balance_colony_b
				change_building_level_in_location = {
					building = scope:balance_colony_t
					value = local_var:balance_colony_lvl
					owner = scope:balance_colony_o
				}
				if = {
					limit = { has_local_variable = balance_colony_was_kept }
					random_buildings_in_location = {
						limit = {
							building_type = scope:balance_colony_t
							owner ?= scope:balance_colony_o
						}
						save_temporary_scope_as = balance_colony_n
						scope:balance_colony_here = {
							add_to_variable_list = { name = balance_keep_buildings target = scope:balance_colony_n }
						}
					}
				}
			}
		}
		clear_variable_list = balance_colony_reopen
	}
}

# every day, from balance_colony_subsidy_renew (root = the overlord): the
# locked buildings in our subjects' lands that were closed
balance_colony_reopen_all_kept = {
	every_subject = {
		every_owned_location = {
			limit = {
				OR = {
					has_variable = balance_keep_all
					has_variable_list = balance_keep_buildings
				}
			}
			balance_colony_reopen_kept_here = yes
		}
	}
}

balance_colony_unsubsidize_here = {
	owner = { save_temporary_scope_as = balance_colony_o }
	every_buildings_in_location = {
		limit = {
			owner ?= scope:balance_colony_o
			is_subsidized = yes
		}
		set_subsidized = no
	}
}

balance_colony_subsidy_on_here = {
	set_variable = { name = balance_subsidy_all value = root }
	save_temporary_scope_as = balance_colony_sl
	root = {
		if = {
			limit = { NOT = { is_target_in_variable_list = { name = balance_subsidy_locations target = scope:balance_colony_sl } } }
			add_to_variable_list = { name = balance_subsidy_locations target = scope:balance_colony_sl }
		}
	}
	balance_colony_reopen_here = yes
	balance_colony_subsidize_here = yes
	root = { balance_colony_subsidy_seed = yes }
}

balance_colony_subsidy_off_here = {
	remove_variable = balance_subsidy_all
	save_temporary_scope_as = balance_colony_sl
	root = { remove_list_variable = { name = balance_subsidy_locations target = scope:balance_colony_sl } }
	balance_colony_unsubsidize_here = yes
}

# ---- the daily renewal (root = the overlord), a hidden event that re-arms
# itself every day while the overlord has subsidised locations
# (events/balance_colony_subsidy_events.txt); the same chain as the combat
# probe's. balance_subsidy_running stops a second chain from starting beside
# a live one; balance_subsidy_alive is set on every tick, so the monthly pulse
# can tell a chain that died without dropping its flag and start a new one.
balance_colony_subsidy_seed = {
	if = {
		limit = {
			NOT = { has_variable = balance_subsidy_running }
			balance_colony_daily_needed = yes
		}
		set_variable = balance_subsidy_running
		set_variable = balance_subsidy_alive
		trigger_event_silently = {
			id = balance_colony_subsidy.1
			days = 1
		}
	}
}

# a building lock was set (root = the overlord): from now on the daily chain
# also reopens locked buildings the subject closed
balance_colony_locks_on = {
	if = {
		limit = { NOT = { has_variable = balance_colony_uses_locks } }
		set_variable = balance_colony_uses_locks
	}
	balance_colony_subsidy_seed = yes
}

# monthly, from on_action/balance_colony_on_action.txt
balance_colony_subsidy_monthly = {
	# locks set before the daily chain reopened buildings carry no flag yet
	if = {
		limit = {
			is_ai = no
			NOT = { has_variable = balance_colony_uses_locks }
			any_subject = {
				any_owned_location = {
					OR = {
						has_variable = balance_keep_all
						has_variable_list = balance_keep_buildings
					}
				}
			}
		}
		set_variable = balance_colony_uses_locks
	}
	if = {
		limit = {
			has_variable = balance_subsidy_running
			NOT = { has_variable = balance_subsidy_alive }
		}
		# no tick for a whole month from a chain that ticks daily: it is gone
		remove_variable = balance_subsidy_running
	}
	if = {
		limit = { has_variable = balance_subsidy_alive }
		remove_variable = balance_subsidy_alive
	}
	balance_colony_subsidy_seed = yes
}

# every day (root = the overlord): locked buildings the subject closed are
# rebuilt open, and "Keep open" locations are renewed. balance_subsidy_locations
# on the overlord spares this a walk over every subject's every location for
# them. A location no longer a subject's lapses: its subsidies stay as they
# are, the new owner decides.
balance_colony_subsidy_renew = {
	if = {
		limit = { has_variable = balance_colony_uses_locks }
		balance_colony_reopen_all_kept = yes
	}
	every_in_list = {
		variable = balance_subsidy_locations
		limit = { var:balance_subsidy_all ?= root }
		if = {
			limit = { owner ?= { is_subject_of = root } }
			balance_colony_reopen_here = yes
			balance_colony_subsidize_here = yes
		}
		else = {
			remove_variable = balance_subsidy_all
		}
	}
	# drop the lapsed ones from the list, but not while walking it
	if = {
		limit = {
			any_in_list = {
				variable = balance_subsidy_locations
				NOT = { var:balance_subsidy_all ?= root }
			}
		}
		clear_variable_list = balance_subsidy_rebuild
		every_in_list = {
			variable = balance_subsidy_locations
			limit = { var:balance_subsidy_all ?= root }
			save_temporary_scope_as = balance_colony_sl
			root = { add_to_variable_list = { name = balance_subsidy_rebuild target = scope:balance_colony_sl } }
		}
		clear_variable_list = balance_subsidy_locations
		every_in_list = {
			variable = balance_subsidy_rebuild
			save_temporary_scope_as = balance_colony_sl
			root = { add_to_variable_list = { name = balance_subsidy_locations target = scope:balance_colony_sl } }
		}
		clear_variable_list = balance_subsidy_rebuild
	}
}

# ---- giving back the buildings the old "can't be closed" lock took
# (removed 2026-09-27: taking a building split it in two whenever the subject
# built more levels of that type). this = the location, root = the overlord.
balance_colony_free_hold_here = {
	save_temporary_scope_as = balance_colony_here
	owner = { save_temporary_scope_as = balance_colony_o }
	every_in_list = {
		variable = balance_hold_buildings
		limit = { owner ?= root }
		save_temporary_scope_as = balance_colony_b
		building_type = { save_temporary_scope_as = balance_colony_t }
		if = {
			limit = {
				scope:balance_colony_here = {
					any_buildings_in_location = {
						building_type = scope:balance_colony_t
						owner ?= scope:balance_colony_o
					}
				}
			}
			# the subject built a second one of this type beside ours:
			# move our levels into theirs, then remove ours
			scope:balance_colony_here = {
				random_buildings_in_location = {
					limit = {
						building_type = scope:balance_colony_t
						owner ?= scope:balance_colony_o
					}
					save_temporary_scope_as = balance_colony_s
				}
			}
			while = {
				limit = { scope:balance_colony_b = { building_level >= 2 } }
				scope:balance_colony_b = { change_building_level = -1 }
				scope:balance_colony_s = { change_building_level = 1 }
			}
			if = {
				limit = { scope:balance_colony_b = { building_level >= 1 } }
				scope:balance_colony_s = { change_building_level = 1 }
			}
			scope:balance_colony_here = { destroy_building = scope:balance_colony_b }
		}
		else = {
			change_building_owner = scope:balance_colony_o
		}
	}
	clear_variable_list = balance_hold_buildings
	if = { limit = { has_variable = balance_hold_all } remove_variable = balance_hold_all }
}

# every such building in our subjects' lands (root = the overlord)
balance_colony_return_all_held = {
	every_subject = {
		every_owned_location = {
			limit = { has_variable_list = balance_hold_buildings }
			balance_colony_free_hold_here = yes
		}
	}
}
