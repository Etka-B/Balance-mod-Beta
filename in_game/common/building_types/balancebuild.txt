REPLACE:bailiff = {
	audio_tier = 2
	is_foreign = no
	max_levels = 1
	pop_type = soldiers
	
	employment_size = bailiff_employment
	category = government_category
	
	rural_settlement = yes
	expensive = yes
	build_time = rural_build_time
	
	forbidden_for_estates = yes
	
	location_potential = {
		owner = {
			OR = {
				this = scope:actor
				is_subject_of = scope:actor
			}
		}
	}
	
	allow = {
		trigger_if = {
			limit = {
				owner = {
					modifier:num_bailiffs <= 0
				}
			}
			custom_tooltip = {
				text = no_local_bailiff_tt
				always = no
			}
		}
		trigger_else = {
			trigger_if = {
				limit = {
					owner = {
						this = scope:actor
					}
				}
				custom_tooltip = {
					text = local_bailiff_tt
					scope:actor = {
						total_building_levels_including_construction:bailiff < modifier:num_bailiffs
					}
				}
			}
			trigger_else = {
				custom_tooltip = {
					text = local_bailiff_subject_tt
					owner = {
						total_building_levels_including_construction:bailiff < modifier:num_bailiffs
					}
				}
			}
		}
	}
	
	owner = {
		OR = {
			this = scope:actor
			is_subject_of = scope:actor
		}
	}

	# Checked on the location (root), not on scope:building: a building still
	# under construction has no owner yet, so the old check was always false
	# and its construction could not be cancelled.
	ai_forbid_shutdown = yes  # Balance mod: no AI ever closes it
	can_destroy = {
		balance_building_not_kept = yes  # Balance mod: colony window building lock

		OR = {
			owner = scope:actor
			owner = {
				is_subject_of = scope:actor
			}
		}
	}
	
	
	unique_production_methods = {
		balance_bailiff_input = {
			paper = 0.1
			tools = 0.15
			weaponry = 0.25
			stone = 0.1
			category = building_maintenance
		}
	}

	modifier = {
		local_max_control = 0.2
		local_nobles_estate_power = 1.0
		local_peasant_enfranchisment = -0.1
	}
	
	
}


REPLACE:local_governor = {
	audio_tier = 2
	is_foreign = no
	pop_type = nobles
	max_levels = 1
	important_for_UI = yes
	
	employment_size = large_noble_employment
	
	category = government_category
	
	rural_settlement = no
	town = no
	city = yes
	megalopolis = yes

	allow = {
		trigger_if = {
			limit = {
				owner = {
					this = scope:actor
				}
			}
			custom_tooltip = {
				text = local_governor_tt
				scope:actor = {
					total_building_levels_including_construction:local_governor < {
						value = {
							value = modifier:num_local_governors
							subtract = total_building_levels_including_construction:lieutenancy
							subtract = total_building_levels_including_construction:balance_colonial_governor  # Balance mod: shares the slot
						}
					}
				}
			}
		}
		trigger_else = {
			custom_tooltip = {
				text = local_governor_subject_tt
				owner = {
					total_building_levels_including_construction:local_governor < {
						value = {
							value = modifier:num_local_governors
							subtract = total_building_levels_including_construction:lieutenancy
							subtract = total_building_levels_including_construction:balance_colonial_governor  # Balance mod: shares the slot
						}
					}
				}
			}
		}
		owner = {
			OR = {
				this = scope:actor
				is_subject_of = scope:actor
			}
		}
		trigger_if = {
			limit = { has_building = building_type:seljuk_mint }
			NOT = { has_building = building_type:seljuk_mint }
		}
	}

	# Checked on the location (root), not on scope:building: a building still
	# under construction has no owner yet, so the old check was always false
	# and its construction could not be cancelled.
	can_destroy = {
		balance_building_not_kept = yes  # Balance mod: colony window building lock

		OR = {
			owner = {
				is_subject_of = scope:actor
			}
			AND = {
				owner = scope:actor
				# a subject cannot take down a governor its overlord has kept
				# (country_interactions/balance_subject_locks.txt)
				trigger_if = {
					limit = {
						has_variable = balance_keep_governor
						owner = { is_subject = yes }
					}
					custom_tooltip = {
						text = balance_keep_governor_tt
						always = no
					}
				}
			}
		}
	}

	location_potential = {
		has_road_to_capital = yes
		owner.capital != this
	}

	remove_if = {
		location = { has_road_to_capital = no }
		NOT = { owner = { at_war = yes } }
		NOT = { owner = { in_civil_war = yes } }
	}

	expensive = yes
	
	automation_build_allowed = no
	ai_forbid_shutdown = yes
	forbidden_for_estates = yes
	important_for_AI = yes
	ai_ignore_maintenance = yes

	build_time = government_build_time


	unique_production_methods = {
		balance_local_governor_maintenance = {
			tools = 1
			masonry = 1
			weaponry = 0.5
			paper = 3
			books = 1
			goods_gold = 1
			furniture = 2
			fine_cloth = 1

			category = building_maintenance
		}
	}


	construction_demand = local_governor_construction

	modifier = {
		local_proximity_source = 80
	}
}
