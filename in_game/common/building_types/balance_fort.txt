
TRY_REPLACE:stockade = {
	audio_tier = 1
	
	max_levels = 1
	pop_type = soldiers
	
	category = defense_category
	
	employment_size = stockade_employment
	
	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes
	
	
	build_time = small_fort_building

	expensive = yes
	
	

	
	unique_production_methods = {
	
		stockade_maintenance = {
			lumber = 0.1
			weaponry = 0.02
			tar = 0.02
			category = building_maintenance
		}
	}
	construction_demand = stockade_construction
	


	allow = {
		NOT = {
			any_neighbor_location = {
				OR = {
					AND = {
						# A neighbour with no owner -- a seazone, a wasteland, anything
						# uncolonised -- has no top_owner, and reading it there is an
						# error the engine logs every single time. The fort build check
						# walks every neighbour of every candidate location, so it fired
						# constantly: 1,098 of them in one 1 MB slice of error.log, and
						# error.log had already rotated five times. Each one is two disk
						# writes (error.log and debug.log, which reached 1.1 GB), which
						# is what turned a script mistake into a frame-rate problem.
						exists = owner
						top_owner = scope:actor
						OR = {
							modifier:fort_level > 0
							modifier:minimum_fort_level > 0
						}
					}
					any_buildings_in_location = {
						building_category = defense_category
						building_levels_under_construction >= 1
					}
				}
			}
		}
	}

	modifier = {
		local_garrison_size = 0.1
		local_unrest = -0.025
		local_nobles_estate_power = 0.5
	}
	
	raw_modifier = {
		fort_level = 1
		local_nobles_desired_pop = 0.005
		pure_tooltip_entry = pte_no_propagating_zone_of_control            # propagating_zone_of_control = no
	}
}
