# THE COLONY WINDOW'S BUTTONS (gui/balance_colony_window.gui). Lists are
# filled by scripted_effects/balance_colony_effects.txt; what was picked is
# kept in balance_colony_subject / balance_colony_location on root, the
# filters in balance_colony_f_rural/_town/_city/_goods.

balance_colony_open = {
	scope = country
	effect = {
		balance_colony_return_all_held = yes
		balance_colony_fill_subjects = yes
		if = {
			limit = {
				has_variable = balance_colony_subject
				NOT = { var:balance_colony_subject ?= { is_subject_of = root } }
			}
			remove_variable = balance_colony_subject
			clear_variable_list = balance_colony_selected
		}
		if = {
			limit = {
				has_variable = balance_colony_location
				NOT = { var:balance_colony_location ?= { owner ?= { is_subject_of = root } } }
			}
			remove_variable = balance_colony_location
		}
		balance_colony_fill_locations = yes
		balance_colony_fill_buildings = yes
	}
}

# from the location window's button: straight to that location
balance_colony_can_open_at = {
	scope = country
	saved_scopes = { location }
	is_shown = { scope:location = { owner ?= { is_subject_of = root } } }
}

balance_colony_open_at = {
	scope = country
	saved_scopes = { location }
	is_valid = { scope:location = { owner ?= { is_subject_of = root } } }
	effect = {
		balance_colony_return_all_held = yes
		balance_colony_fill_subjects = yes
		scope:location = { owner = { save_temporary_scope_as = balance_colony_s2 } }
		if = {
			limit = { NOT = { var:balance_colony_subject ?= scope:balance_colony_s2 } }
			clear_variable_list = balance_colony_selected
		}
		set_variable = { name = balance_colony_subject value = scope:balance_colony_s2 }
		set_variable = { name = balance_colony_location value = scope:location }
		balance_colony_clear_filters = yes
		balance_colony_fill_locations = yes
		balance_colony_fill_buildings = yes
	}
}

balance_colony_pick_subject = {
	scope = country
	saved_scopes = { subject }
	effect = {
		set_variable = { name = balance_colony_subject value = scope:subject }
		if = {
			limit = { has_variable = balance_colony_location }
			remove_variable = balance_colony_location
		}
		clear_variable_list = balance_colony_selected
		if = {
			limit = { has_variable = balance_colony_f_goods }
			remove_variable = balance_colony_f_goods
		}
		balance_colony_fill_locations = yes
		balance_colony_fill_buildings = yes
	}
}

# ---- filters
balance_colony_filter_all = {
	scope = country
	effect = {
		if = { limit = { has_variable = balance_colony_f_rural } remove_variable = balance_colony_f_rural }
		if = { limit = { has_variable = balance_colony_f_town } remove_variable = balance_colony_f_town }
		if = { limit = { has_variable = balance_colony_f_city } remove_variable = balance_colony_f_city }
		balance_colony_fill_locations = yes
	}
}

balance_colony_filter_all_on = {
	scope = country
	is_shown = {
		NOT = { has_variable = balance_colony_f_rural }
		NOT = { has_variable = balance_colony_f_town }
		NOT = { has_variable = balance_colony_f_city }
	}
}

# ---- selection for the bulk buttons
balance_colony_toggle_select = {
	scope = country
	saved_scopes = { location }
	effect = {
		if = {
			limit = { is_target_in_variable_list = { name = balance_colony_selected target = scope:location } }
			remove_list_variable = { name = balance_colony_selected target = scope:location }
		}
		else = {
			add_to_variable_list = { name = balance_colony_selected target = scope:location }
		}
	}
}

balance_colony_is_selected = {
	scope = country
	saved_scopes = { location }
	is_shown = { is_target_in_variable_list = { name = balance_colony_selected target = scope:location } }
}

balance_colony_select_shown = {
	scope = country
	effect = {
		every_in_list = {
			variable = balance_colony_locations
			save_temporary_scope_as = balance_colony_l
			root = {
				if = {
					limit = { NOT = { is_target_in_variable_list = { name = balance_colony_selected target = scope:balance_colony_l } } }
					add_to_variable_list = { name = balance_colony_selected target = scope:balance_colony_l }
				}
			}
		}
	}
}

balance_colony_select_clear = {
	scope = country
	effect = { clear_variable_list = balance_colony_selected }
}

# ---- bulk, on every selected location: each button turns one thing on
# everywhere in the selection, or, when it is on everywhere, off
balance_colony_sel_town = {
	scope = country
	effect = {
		if = {
			limit = {
				any_in_list = {
					variable = balance_colony_selected
					owner ?= { is_subject_of = root }
					has_variable = balance_no_town
				}
				NOT = {
					any_in_list = {
						variable = balance_colony_selected
						owner ?= { is_subject_of = root }
						location_rank = location_rank:rural_settlement
						NOT = { has_variable = balance_no_town }
					}
				}
			}
			every_in_list = {
				variable = balance_colony_selected
				limit = {
					owner ?= { is_subject_of = root }
					has_variable = balance_no_town
				}
				remove_variable = balance_no_town
			}
		}
		else = {
			every_in_list = {
				variable = balance_colony_selected
				limit = {
					owner ?= { is_subject_of = root }
					location_rank = location_rank:rural_settlement
					NOT = { has_variable = balance_no_town }
				}
				set_variable = balance_no_town
			}
		}
	}
}

# select every shown location, or, when anything is selected, clear the selection
balance_colony_select_toggle = {
	scope = country
	effect = {
		if = {
			limit = { any_in_list = { variable = balance_colony_selected always = yes } }
			clear_variable_list = balance_colony_selected
		}
		else = {
			every_in_list = {
				variable = balance_colony_locations
				save_temporary_scope_as = balance_colony_l
				root = {
					add_to_variable_list = { name = balance_colony_selected target = scope:balance_colony_l }
				}
			}
		}
	}
}

# ---- the building locks (scripted_triggers/balance_colony_triggers.txt)
balance_colony_is_kept = {
	scope = country
	saved_scopes = { location building }
	is_shown = {
		scope:location = {
			OR = {
				has_variable = balance_keep_all
				is_target_in_variable_list = { name = balance_keep_buildings target = scope:building }
			}
		}
	}
}

# a building the subject may not take down
balance_colony_toggle_keep = {
	scope = country
	saved_scopes = { location building }
	is_valid = {
		scope:location = { owner ?= { is_subject_of = root } }
		custom_tooltip = {
			text = balance_colony_lock_unsupported_tt
			scope:building = { NOT = { balance_building_lock_unsupported = yes } }
		}
		custom_tooltip = {
			text = balance_colony_keep_all_tt
			scope:location = { NOT = { has_variable = balance_keep_all } }
		}
	}
	effect = {
		scope:location = {
			if = {
				limit = { is_target_in_variable_list = { name = balance_keep_buildings target = scope:building } }
				remove_list_variable = { name = balance_keep_buildings target = scope:building }
			}
			else = {
				add_to_variable_list = { name = balance_keep_buildings target = scope:building }
			}
		}
	}
}

# the whole picked location at once
balance_colony_keep_here = {
	scope = country
	saved_scopes = { location }
	is_valid = { scope:location = { owner ?= { is_subject_of = root } } }
	effect = {
		scope:location = {
			if = {
				limit = { has_variable = balance_keep_all }
				balance_colony_free_keep_here = yes
			}
			else = {
				balance_colony_keep_buildings_here = yes
			}
		}
		balance_colony_fill_buildings = yes
	}
}

balance_colony_keep_here_on = {
	scope = country
	saved_scopes = { location }
	is_shown = { scope:location = { has_variable = balance_keep_all } }
}

# ---- subsidies (scripted_effects/balance_colony_effects.txt): the subject
# pays for its buildings in the picked location, so its AI has no loss to
# close them over
balance_colony_subsidy_here = {
	scope = country
	saved_scopes = { location }
	is_valid = { scope:location = { owner ?= { is_subject_of = root } } }
	effect = {
		scope:location = {
			if = {
				limit = { var:balance_subsidy_all ?= root }
				balance_colony_subsidy_off_here = yes
			}
			else = {
				balance_colony_subsidy_on_here = yes
			}
		}
	}
}

balance_colony_subsidy_here_on = {
	scope = country
	saved_scopes = { location }
	is_shown = { scope:location = { var:balance_subsidy_all ?= root } }
}
