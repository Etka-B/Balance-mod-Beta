# ---- the building filter
balance_colony_filter_btype = {
	scope = country
	saved_scopes = { btype }
	effect = {
		if = {
			limit = { var:balance_colony_f_btype ?= scope:btype }
			remove_variable = balance_colony_f_btype
		}
		else = {
			set_variable = { name = balance_colony_f_btype value = scope:btype }
		}
		balance_colony_fill_locations = yes
	}
}

balance_colony_is_btype = {
	scope = country
	saved_scopes = { btype }
	is_shown = { var:balance_colony_f_btype ?= scope:btype }
}

# ---- subsidies for the selection: only the filtered building type when one
# is picked, else every building of the subject's there
balance_colony_sel_subsidy = {
	scope = country
	is_valid = { any_in_list = { variable = balance_colony_selected owner ?= { is_subject_of = root } } }
	effect = {
		if = {
			limit = { has_variable = balance_colony_f_btype }
			var:balance_colony_f_btype = { save_temporary_scope_as = balance_colony_st }
			if = {
				limit = {
					NOT = {
						any_in_list = {
							variable = balance_colony_selected
							owner ?= { is_subject_of = root }
							NOT = { is_target_in_variable_list = { name = balance_subsidy_types target = scope:balance_colony_st } }
						}
					}
				}
				every_in_list = {
					variable = balance_colony_selected
					limit = { owner ?= { is_subject_of = root } }
					balance_colony_subsidy_type_off_here = yes
				}
			}
			else = {
				every_in_list = {
					variable = balance_colony_selected
					limit = { owner ?= { is_subject_of = root } }
					balance_colony_subsidy_type_on_here = yes
				}
			}
		}
		else = {
			if = {
				limit = {
					NOT = {
						any_in_list = {
							variable = balance_colony_selected
							owner ?= { is_subject_of = root }
							NOT = { var:balance_subsidy_all ?= root }
						}
					}
				}
				every_in_list = {
					variable = balance_colony_selected
					limit = { owner ?= { is_subject_of = root } }
					balance_colony_subsidy_off_here = yes
				}
			}
			else = {
				every_in_list = {
					variable = balance_colony_selected
					limit = {
						owner ?= { is_subject_of = root }
						NOT = { var:balance_subsidy_all ?= root }
					}
					balance_colony_subsidy_on_here = yes
				}
			}
		}
		balance_colony_fill_buildings = yes
	}
}

balance_colony_sel_subsidy_on = {
	scope = country
	is_shown = {
		any_in_list = { variable = balance_colony_selected owner ?= { is_subject_of = root } }
		NOT = {
			any_in_list = {
				variable = balance_colony_selected
				owner ?= { is_subject_of = root }
				NOT = { var:balance_subsidy_all ?= root }
			}
		}
	}
}

balance_prio_toggle_b = {
	scope = country
	saved_scopes = { btype }
	is_valid = {
		has_variable = balance_colony_subject
		var:balance_colony_subject ?= { is_subject_of = root }
		var:balance_colony_subject = {
			OR = {
				balance_prio_has_b = yes
				balance_prio_slot_empty = { S = 20 }
			}
		}
	}
	effect = {
		balance_tier_clear = yes	# the level popup belonged to the rows as they were
		var:balance_colony_subject = {
			if = {
				limit = { balance_prio_has_b = yes }
				balance_prio_remove = { K = b T = btype }
			}
			else = {
				balance_prio_add = { K = b T = btype }
				if = {
					limit = { is_target_in_variable_list = { name = balance_forbid_buildings target = scope:btype } }
					remove_list_variable = { name = balance_forbid_buildings target = scope:btype }
				}
			}
		}
	}
}

balance_prio_save_sel = {
	scope = country
	is_valid = {
		has_variable = balance_colony_subject
		var:balance_colony_subject ?= { is_subject_of = root }
	}
	effect = {
		var:balance_colony_subject = { save_temporary_scope_as = balance_prio_s }
		scope:balance_prio_s = {
			every_in_list = {
				variable = balance_prio_locs
				limit = { var:balance_prio_pick ?= scope:balance_prio_s }
				remove_variable = balance_prio_pick
			}
			clear_variable_list = balance_prio_locs
		}
		every_in_list = {
			variable = balance_colony_selected
			limit = { owner ?= scope:balance_prio_s }
			set_variable = { name = balance_prio_pick value = scope:balance_prio_s }
			save_temporary_scope_as = balance_prio_l
			scope:balance_prio_s = { add_to_variable_list = { name = balance_prio_locs target = scope:balance_prio_l } }
		}
	}
}

balance_prio_toggle_loc = {
	scope = country
	saved_scopes = { location }
	is_valid = {
		has_variable = balance_colony_subject
		var:balance_colony_subject ?= { is_subject_of = root }
		scope:location = { owner ?= root.var:balance_colony_subject }
	}
	effect = {
		var:balance_colony_subject = { save_temporary_scope_as = balance_prio_s }
		scope:location = {
			if = {
				limit = { var:balance_prio_pick ?= scope:balance_prio_s }
				remove_variable = balance_prio_pick
				save_temporary_scope_as = balance_prio_l
				scope:balance_prio_s = { remove_list_variable = { name = balance_prio_locs target = scope:balance_prio_l } }
			}
			else = {
				set_variable = { name = balance_prio_pick value = scope:balance_prio_s }
				save_temporary_scope_as = balance_prio_l
				scope:balance_prio_s = { add_to_variable_list = { name = balance_prio_locs target = scope:balance_prio_l } }
			}
		}
	}
}
