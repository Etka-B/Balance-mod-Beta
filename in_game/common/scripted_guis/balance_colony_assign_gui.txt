# THE COLONY WINDOW'S "COLONY ASSIGNMENT" TAB (colony_assign_v7.py); effects in
# scripted_effects/balance_colony_assign_effects.txt. scope:node = a region,
# area, province (definition) or location; the suffix says which.

balance_assign_refresh = {
	scope = country
	effect = {
		balance_colony_fill_subjects = yes
		balance_assign_fill = yes
	}
}

balance_assign_home = {
	scope = country
	effect = {
		if = { limit = { has_variable = balance_assign_nav_r } remove_variable = balance_assign_nav_r }
		if = { limit = { has_variable = balance_assign_nav_a } remove_variable = balance_assign_nav_a }
		if = { limit = { has_variable = balance_assign_nav_p } remove_variable = balance_assign_nav_p }
		if = { limit = { has_variable = balance_assign_edit } remove_variable = balance_assign_edit }
		if = { limit = { has_variable = balance_assign_edit_r } remove_variable = balance_assign_edit_r }
		if = { limit = { has_variable = balance_assign_edit_a } remove_variable = balance_assign_edit_a }
		if = { limit = { has_variable = balance_assign_edit_p } remove_variable = balance_assign_edit_p }
		if = { limit = { has_variable = balance_assign_edit_l } remove_variable = balance_assign_edit_l }
		balance_assign_fill_children = yes
	}
}

balance_assign_open_r = {
	scope = country
	saved_scopes = { node area }
	effect = {
		set_variable = { name = balance_assign_nav_r value = scope:node }
		if = { limit = { has_variable = balance_assign_nav_a } remove_variable = balance_assign_nav_a }
		if = { limit = { has_variable = balance_assign_nav_p } remove_variable = balance_assign_nav_p }
		balance_assign_set_edit = { K = r }
		if = { limit = { exists = scope:area } set_variable = { name = balance_assign_focus_area value = scope:area } }
		balance_assign_fill_children = yes
	}
}

balance_assign_focus_r = {
	scope = country
	saved_scopes = { node }
	effect = {
		scope:node = {
			ordered_location_in_region = {
				order_by = population
				max = 1
				check_range_bounds = no
				save_temporary_scope_as = balance_assign_fl
			}
		}
		if = {
			limit = { exists = scope:balance_assign_fl }
			set_variable = { name = balance_assign_focus value = scope:balance_assign_fl }
		}
	}
}
